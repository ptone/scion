// PROBE (ca-msg-arch, B5/F1 re-review of 24b97149).
//
// F2 was reported as "display only". It is not. The unconditional override
// sets Sender = "agent:" + agentIdent.ID() (a UUID). MessageBrokerProxy's
// broadcast fan-out still skips the sender by SLUG:
//
//	messagebroker.go:713   if msg.Sender == "agent:"+agent.Slug { continue }
//
// A UUID never equals a slug, so the comparison stops matching and the
// broadcasting agent receives its own broadcast. handleProjectBroadcast's
// own loop was fixed to skip on authID, but that loop only feeds the
// direct-dispatch fallback and the response counts; when a broker proxy is
// present the skip is re-derived from Sender inside fanOutToProject.
//
// Asserts the invariant "a broadcasting agent is not delivered its own
// broadcast". Expected to FAIL on 24b97149.
func TestProbe_B5F1_BroadcastingAgentReceivesItsOwnBroadcast(t *testing.T) {
	srv, s := testServer(t)
	ctx := context.Background()

	project := &store.Project{
		ID: api.NewUUID(), Name: "selfskip", Slug: "selfskip",
		Visibility: store.VisibilityPrivate,
	}
	if err := s.CreateProject(ctx, project); err != nil {
		t.Fatalf("CreateProject: %v", err)
	}
	sender := &store.Agent{
		ID: api.NewUUID(), Name: "sender-agent", Slug: "sender-agent",
		ProjectID: project.ID, Phase: "running", Visibility: store.VisibilityPrivate,
		RuntimeBrokerID: "probe-broker",
	}
	peer := &store.Agent{
		ID: api.NewUUID(), Name: "peer-agent", Slug: "peer-agent",
		ProjectID: project.ID, Phase: "running", Visibility: store.VisibilityPrivate,
		RuntimeBrokerID: "probe-broker",
	}
	if err := s.CreateAgent(ctx, sender); err != nil {
		t.Fatalf("CreateAgent sender: %v", err)
	}
	if err := s.CreateAgent(ctx, peer); err != nil {
		t.Fatalf("CreateAgent peer: %v", err)
	}

	inproc := eventbus.NewInProcessEventBus(slog.Default())
	fanout := eventbus.NewFanOutEventBus([]eventbus.NamedEventBus{
		{Name: eventbus.InProcessBusName, Bus: inproc},
		{Name: "web", Bus: nullSpokeEventBus{}},
	}, slog.Default())
	events := NewChannelEventPublisher()
	defer events.Close()
	proxy := NewMessageBrokerProxy(fanout, s, events, func() AgentDispatcher { return noopDispatcher{} }, slog.Default())
	proxy.Start()
	t.Cleanup(proxy.Stop)
	srv.SetMessageBrokerProxy(proxy)
	proxy.subscribeProjectBroadcast(project.ID)
	proxy.subscribeAgent(project.ID, sender.Slug)
	proxy.subscribeAgent(project.ID, peer.Slug)

	body, _ := json.Marshal(BroadcastMessageRequest{
		StructuredMessage: &messages.StructuredMessage{
			Version:   messages.Version,
			Timestamp: time.Now().UTC().Format(time.RFC3339),
			Type:      messages.TypeInstruction,
			Sender:    "agent:sender-agent",
			Msg:       "hello project",
		},
	})
	req := httptest.NewRequest(http.MethodPost, "/api/v1/projects/"+project.ID+"/broadcast", bytes.NewReader(body))
	req.Header.Set("Content-Type", "application/json")
	// Authenticate AS the sending agent.
	req = req.WithContext(contextWithIdentity(req.Context(),
		&agentIdentityWrapper{&AgentTokenClaims{
			Claims:    jwt.Claims{Subject: sender.ID},
			ProjectID: project.ID,
			Scopes:    []AgentTokenScope{ScopeAgentLifecycle},
		}}))
	rr := httptest.NewRecorder()
	srv.handleProjectBroadcast(rr, req, project.ID)
	t.Logf("broadcast response: %d %s", rr.Code, rr.Body.String())

	countFor := func(agentID string) int {
		res, err := s.ListMessages(ctx, store.MessageFilter{AgentID: agentID}, store.ListOptions{})
		if err != nil {
			t.Fatalf("ListMessages: %v", err)
		}
		return len(res.Items)
	}

	deadline := time.Now().Add(3 * time.Second)
	for time.Now().Before(deadline) {
		if countFor(peer.ID) > 0 {
			break
		}
		time.Sleep(50 * time.Millisecond)
	}
	peerN, selfN := countFor(peer.ID), countFor(sender.ID)
	t.Logf("delivered rows: peer=%d self=%d", peerN, selfN)

	// POSITIVE CONTROL: the peer must have received it, or the probe never
	// exercised the fan-out and the self-count means nothing.
	if peerN == 0 {
		t.Fatalf("PROBE INCONCLUSIVE: the peer agent received nothing, so fan-out never ran")
	}
	if selfN > 0 {
		t.Errorf("SELF-DELIVERY: the broadcasting agent received its own broadcast (%d rows). "+
			"fanOutToProject skips by slug (messagebroker.go:713) but Sender is now the UUID form.", selfN)
	}
}
