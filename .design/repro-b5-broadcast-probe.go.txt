// PROBE (ca-msg-arch review of B5 @ 9241f86f).
//
// B5 fixed handleAgentMessage. handleProjectBroadcast is a SIBLING ingress in
// the same file that still backfills the sender CONDITIONALLY:
//
//	if req.StructuredMessage.Sender == "" { ...derive from auth... }
//
// A caller who supplies a non-empty Sender keeps their own SenderID verbatim.
// Nothing server-side sets Broadcasted on the structured message, so the
// message reaches MessageBrokerProxy.deliverToAgent with Broadcasted=false,
// where the `if !msg.Broadcasted` dual-write mints a DM conversation keyed on
// the attacker-chosen SenderID -- for EVERY running agent in the project.
//
// Asserts the security invariant. Expected to FAIL on 9241f86f.
func TestProbe_B5_BroadcastIngressStillAcceptsSpoofedSender(t *testing.T) {
	srv, s := testServer(t)
	ctx := context.Background()

	project := &store.Project{
		ID: api.NewUUID(), Name: "b5-bcast", Slug: "b5-bcast",
		Visibility: store.VisibilityPrivate,
	}
	if err := s.CreateProject(ctx, project); err != nil {
		t.Fatalf("CreateProject: %v", err)
	}
	attacker := &store.User{ID: api.NewUUID(), Email: "attacker@example.com", DisplayName: "Attacker", Role: store.UserRoleMember, Status: "active"}
	victim := &store.User{ID: api.NewUUID(), Email: "victim@example.com", DisplayName: "Victim", Role: store.UserRoleMember, Status: "active"}
	if err := s.CreateUser(ctx, attacker); err != nil {
		t.Fatalf("CreateUser attacker: %v", err)
	}
	if err := s.CreateUser(ctx, victim); err != nil {
		t.Fatalf("CreateUser victim: %v", err)
	}
	agent := &store.Agent{
		ID: api.NewUUID(), Name: "target-agent", Slug: "target-agent",
		ProjectID: project.ID, Phase: "running", Visibility: store.VisibilityPrivate,
		RuntimeBrokerID: "probe-broker",
	}
	if err := s.CreateAgent(ctx, agent); err != nil {
		t.Fatalf("CreateAgent: %v", err)
	}

	inproc := eventbus.NewInProcessEventBus(slog.Default())
	fanout := eventbus.NewFanOutEventBus([]eventbus.NamedEventBus{
		{Name: eventbus.InProcessBusName, Bus: inproc},
		{Name: "web", Bus: nullSpokeEventBus{}},
	}, slog.Default())
	events := NewChannelEventPublisher()
	defer events.Close()
	proxy := NewMessageBrokerProxy(fanout, s, events, func() AgentDispatcher { return noopDispatcher{} }, slog.Default())
	proxy.Start()
	t.Cleanup(proxy.Stop)
	srv.SetMessageBrokerProxy(proxy)
	proxy.subscribeProjectBroadcast(project.ID)
	proxy.subscribeAgent(project.ID, agent.Slug)

	// Attacker authenticates as themself, but claims to be the victim.
	// Sender is NON-EMPTY, so handleProjectBroadcast's conditional backfill
	// is skipped and SenderID survives untouched.
	spoofed := &messages.StructuredMessage{
		Version:     messages.Version,
		Timestamp:   time.Now().UTC().Format(time.RFC3339),
		Type:        messages.TypeInstruction,
		Sender:      "user:" + victim.Email,
		SenderID:    victim.ID,
		Msg:         "broadcast as the victim",
		Broadcasted: false, // never forced true server-side
	}
	body, _ := json.Marshal(BroadcastMessageRequest{StructuredMessage: spoofed})
	req := httptest.NewRequest(http.MethodPost, "/api/v1/projects/"+project.ID+"/broadcast", bytes.NewReader(body))
	req.Header.Set("Content-Type", "application/json")
	req = req.WithContext(contextWithIdentity(req.Context(),
		NewAuthenticatedUser(attacker.ID, attacker.Email, attacker.DisplayName, "user", "web")))
	rr := httptest.NewRecorder()
	srv.handleProjectBroadcast(rr, req, project.ID)
	t.Logf("broadcast response: %d %s", rr.Code, rr.Body.String())

	// Give the async bus fan-out a moment to land.
	deadline := time.Now().Add(3 * time.Second)
	spoofedKey, err := messages.DMConversationKey("user", victim.ID, "agent", agent.ID)
	if err != nil {
		t.Fatalf("DMConversationKey: %v", err)
	}
	var found bool
	for time.Now().Before(deadline) {
		if c, e := s.GetConversationByExternalRef(ctx, "native", spoofedKey); e == nil && c != nil {
			found = true
			t.Logf("conversation minted under spoofed key: id=%s ref=%s", c.ID, c.ExternalRef)
			break
		}
		time.Sleep(50 * time.Millisecond)
	}
	// POSITIVE CONTROL: the honest key must be minted by this same code path.
	// If neither key appears, the probe never reached the dual-write and a
	// negative result on the spoofed key means nothing.
	honestKey, herr := messages.DMConversationKey("user", attacker.ID, "agent", agent.ID)
	if herr != nil {
		t.Fatalf("DMConversationKey (honest): %v", herr)
	}
	honest := false
	for time.Now().Before(time.Now().Add(2 * time.Second)) {
		if c, e := s.GetConversationByExternalRef(ctx, "native", honestKey); e == nil && c != nil {
			honest = true
			t.Logf("POSITIVE CONTROL: honest-key conversation exists id=%s ref=%s", c.ID, c.ExternalRef)
			break
		}
		time.Sleep(50 * time.Millisecond)
		break
	}

	if found {
		t.Errorf("SECURITY VIOLATION (B5 gap): the project-broadcast ingress minted DM conversation %s "+
			"from an attacker-supplied SenderID. B5 fixed handleAgentMessage only; "+
			"handleProjectBroadcast still backfills the sender conditionally.", spoofedKey)
	} else if !honest {
		t.Fatalf("PROBE INCONCLUSIVE: neither the spoofed key nor the honest key %s was minted — "+
			"the dual-write in deliverToAgent was never reached, so the negative result is meaningless", honestKey)
	} else {
		t.Logf("no conversation under spoofed key %s (positive control passed)", spoofedKey)
	}
}

var _ = http.MethodPost
